cmake_minimum_required(VERSION 3.24)
project(SuiteUiStage3 LANGUAGES CXX)
include(CTest)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Test Svg Network Xml)
find_package(SuiteApp CONFIG REQUIRED)
set(CMAKE_CXX_STANDARD 20)
set(BUILD_SHARED_LIBS OFF)
add_subdirectory("${ZEROSLACK_SOURCE}/thirdparty/qlementine" qlementine EXCLUDE_FROM_ALL)
add_subdirectory("${REGMAP_SOURCE}" regmap)
get_target_property(original regmap_workbench_ui SOURCES)
set(replacement)
foreach(source IN LISTS original)
    if(source STREQUAL "workbench_theme.cpp")
        list(APPEND replacement "${CMAKE_CURRENT_SOURCE_DIR}/workbench_theme.cpp")
    elseif(source STREQUAL "main_window.cpp")
        list(APPEND replacement "${CMAKE_CURRENT_SOURCE_DIR}/regmap_main_window.cpp")
    elseif(IS_ABSOLUTE "${source}" OR source MATCHES "^\\$<")
        list(APPEND replacement "${source}")
    else()
        list(APPEND replacement "${REGMAP_SOURCE}/src/app/${source}")
    endif()
endforeach()
qt_add_executable(regmap_contract_test regmap_contract_test.cpp)
set_target_properties(regmap_contract_test PROPERTIES AUTOMOC ON)
target_link_libraries(regmap_contract_test PRIVATE RegMap::WorkbenchUi Qt6::Test)
target_compile_definitions(regmap_contract_test PRIVATE
    REGMAP_SNAPSHOT_SOURCE="${REGMAP_SOURCE}/tests/ui_snapshot.cpp")
qt_add_executable(regmap_native_review regmap_contract_test.cpp)
set_target_properties(regmap_native_review PROPERTIES AUTOMOC ON)
target_link_libraries(regmap_native_review PRIVATE RegMap::WorkbenchUi Qt6::Test)
target_compile_definitions(regmap_native_review PRIVATE PILOT_NATIVE_REVIEW_DEFAULT
    REGMAP_SNAPSHOT_SOURCE="${REGMAP_SOURCE}/tests/ui_snapshot.cpp")
set_property(TARGET regmap_workbench_ui PROPERTY SOURCES "${replacement}")
target_sources(regmap_workbench_ui PRIVATE regmap_pilot.cpp)
target_link_libraries(regmap_workbench_ui PRIVATE qlementine::qlementine)
target_compile_definitions(regmap_gui_tests PRIVATE PILOT_UI_INITIALIZATION)
set_property(TARGET regmap_gui_tests PROPERTY SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/gui_smoke_tests.cpp")
get_directory_property(regmap_version DIRECTORY "${REGMAP_SOURCE}" DEFINITION PROJECT_VERSION)
target_compile_definitions(regmap_gui_tests PRIVATE REGMAP_EXPECTED_VERSION="${regmap_version}")
foreach(target regmap_gui_tests regmap_core_tests regmap_cli_tests regmap_suiteapp_integration_test)
    if(TARGET ${target})
        target_sources(${target} PRIVATE profile.cpp)
    endif()
endforeach()
